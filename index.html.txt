<script>

// SCREEN NAVIGATION

const badgeTitles = {
1:"MADE WITH SO MUCH LOVE",
2:"BIRTHDAY GREETING",
3:"OUR MEMORY ALBUM",
4:"THREE SWEET WISHES",
5:"A HEARTFELT LETTER"
};

function nextScreen(num) {

document.querySelectorAll('.screen').forEach(screen => {
screen.classList.remove('active');
});

document.getElementById('screen-'+num).classList.add('active');

document.getElementById('topBadgeText').textContent = badgeTitles[num];

window.scrollTo({top:0,behavior:'smooth'});

if(num===5) celebrateAgain();

}


// MUSIC PLAYER

const bgSong = document.getElementById('bgSong');

const playIcon = document.getElementById('playIcon');

const progressBar = document.getElementById('musicProgressBar');

const currentTimeLabel = document.getElementById('currentTimeLabel');

const durationLabel = document.getElementById('durationLabel');


function startMusicPlayback() {

bgSong.play().catch(error => {
console.log("Music playback failed:",error);
});

}

function toggleMusic() {

if(bgSong.paused) {
startMusicPlayback();
} else {
bgSong.pause();
}

}

bgSong.addEventListener('play',()=>{
playIcon.textContent='❚❚';
});

bgSong.addEventListener('pause',()=>{
playIcon.textContent='▶';
});

bgSong.addEventListener('loadedmetadata',()=>{
durationLabel.textContent=formatTime(bgSong.duration);
});

bgSong.addEventListener('timeupdate',()=>{

if(!bgSong.duration) return;

let progress=(bgSong.currentTime/bgSong.duration)*100;

progressBar.style.width=progress+'%';

currentTimeLabel.textContent=formatTime(bgSong.currentTime);

});

function formatTime(time) {

if(!Number.isFinite(time)) return '0:00';

let minutes=Math.floor(time/60);

let seconds=Math.floor(time%60);

return minutes+':'+String(seconds).padStart(2,'0');

}

function seekSong(event) {

if(!bgSong.duration) return;

const rect=event.currentTarget.getBoundingClientRect();

const percentage=(event.clientX-rect.left)/rect.width;

bgSong.currentTime=percentage*bgSong.duration;

}


// ENVELOPE OPEN

function openEnvelope() {

confetti({
particleCount:100,
spread:80,
origin:{y:0.6},
colors:['#f7d6d8','#b34d5d','#facc15','#fb7185']
});

startMusicPlayback();

setTimeout(()=>{
nextScreen(2);
},500);

}

let currentMemoryIdx=0;

function updateMemoryCard() {

const mem=memories[currentMemoryIdx];

document.getElementById('memoryTopTag').textContent=mem.tag;

document.getElementById('memoryImg').src=mem.img;

document.getElementById('memoryStickerText').textContent=mem.title;

document.getElementById('memoryCaption').textContent=mem.caption;

'MEMORY 0'+(currentMemoryIdx+1)+' / 03';

document.getElementById('dot-'+i).className=
i===currentMemoryIdx
?'w-4 h-2 rounded-full bg-rose-400'
:'w-2 h-2 rounded-full bg-stone-300';

}

function nextMemory() {

currentMemoryIdx=(currentMemoryIdx+1)%memories.length;

updateMemoryCard();

}

function prevMemory() {

currentMemoryIdx=(currentMemoryIdx-1+memories.length)%memories.length;

updateMemoryCard();

}


// FLOWER WISHES

function flipWish(element) {

element.classList.toggle('flipped');

confetti({
particleCount:20,
spread:40,
origin:{y:0.7},
colors:['#fda4af','#fcd34d','#e9d5ff']
});

}


// CONFETTI

function celebrateAgain() {

const count=200;

const defaults={origin:{y:0.7}};

function fire(ratio,opts) {

confetti(Object.assign({},defaults,opts,{
particleCount:Math.floor(count*ratio)
}));

}

fire(0.25,{spread:26,startVelocity:55});

fire(0.2,{spread:60});

fire(0.35,{spread:100,decay:0.91,scalar:0.8});

fire(0.1,{spread:120,startVelocity:25,decay:0.92,scalar:1.2});

fire(0.1,{spread:120,startVelocity:45});

}
